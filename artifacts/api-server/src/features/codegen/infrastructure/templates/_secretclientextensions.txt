using Azure.Security.KeyVault.Secrets;
using Microsoft.Extensions.Configuration;

namespace {{NAMESPACE}}.Shared.Extensions
{
    /// <summary>
    /// Resolves <c>@KeyVault:SecretName</c> placeholders inside the loaded
    /// configuration to real secret values from Azure Key Vault.
    /// </summary>
    public static class SecretClientExtensions
    {
        private const string Prefix = "@KeyVault:";

        public static void ResolveAppSettings(this SecretClient client, IConfiguration configuration)
        {
            foreach (var kvp in configuration.AsEnumerable())
            {
                if (kvp.Value == null) continue;
                if (!kvp.Value.StartsWith(Prefix, System.StringComparison.Ordinal)) continue;

                var secretName = kvp.Value.Substring(Prefix.Length);
                try
                {
                    var response = client.GetSecret(secretName);
                    configuration[kvp.Key] = response.Value.Value;
                }
                catch (System.Exception ex)
                {
                    // Surface the failure so a misnamed/missing secret doesn't
                    // silently leave the @KeyVault: placeholder in place and
                    // produce a confusing downstream error. We continue the loop
                    // so other secrets still resolve, and we don't rethrow so
                    // callers don't crash mid-config-load.
                    System.Console.Error.WriteLine(
                        $"[KeyVault] Failed to resolve secret '{secretName}' for config key '{kvp.Key}': {ex.GetType().Name}: {ex.Message}");
                }
            }
        }
    }
}
